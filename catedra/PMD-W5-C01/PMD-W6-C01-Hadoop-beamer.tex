\section{HDFS}
% 11. Original p. 11 (blocks).
\begin{frame}{HDFS: bloques y réplicas}{Distribuir archivos grandes sobre discos locales}
\begin{itemize}[<+->]\setlength\itemsep{5pt}
\item HDFS usa los sistemas de archivos locales de los DataNodes, como ext4 o XFS.
\item Los bloques son grandes y configurables; 128 MB es un tamaño habitual.
\item La replicación mantiene varias copias de cada bloque; el ejemplo usa tres.
\end{itemize}

\section{MapReduce}
% 14. Original p. 12.
\begin{frame}[fragile]{MapReduce: funciones sobre pares clave/valor}{Tareas independientes y combinación de resultados}
\begin{columns}[T,onlytextwidth]
\column{.52\textwidth}
\begin{itemize}[<+->]\setlength\itemsep{7pt}
\item \textbf{Map} transforma cada registro en cero o más pares intermedios.
\item \textbf{Reduce} procesa los valores asociados a una misma clave.
\item El modelo favorece funciones que dependen de sus entradas y evitan estado compartido entre tareas.
\end{itemize}
\column{.44\textwidth}
\begin{lstlisting}
def map(key, value):
    # Emit zero or more pairs
    emit(intermediate_key,
         intermediate_value)
\end{lstlisting}
\begin{uncoverenv}<2->
\begin{lstlisting}
def reduce(key, values):
    # Aggregate values for this key
    emit(key, result)
\end{lstlisting}
\end{uncoverenv}
\smallnote{Pseudocódigo: emit representa la emisión de un par.}
\end{columns}
\begin{block}<3->{La clave coordina el cálculo}
Todos los valores de una clave deben llegar al mismo reductor.
\end{block}
\end{frame}

% 15. Original p. 13.
\begin{frame}{Las fases de MapReduce}{Entrada, map, shuffle y sort, reduce, salida}
\centering
\begin{tikzpicture}[x=2.65cm]
\foreach \i/\lab/\step in {0/Entrada/1,1/Map/2,2/Shuffle + sort/3,3/Reduce/4,4/Salida/4}{
 \node[box,minimum width=23mm,font=\scriptsize,visible on=<\step->] (p\i) at (\i,0) {\lab};}
\foreach \a/\b/\step in {0/1/2,1/2/3,2/3/4,3/4/4}{\draw[flow,visible on=<\step->] (p\a)--(p\b);}
\end{tikzpicture}
\vspace{6mm}
\begin{enumerate}\setlength\itemsep{6pt}
\item<1-> Leer registros desde la entrada, por ejemplo archivos en HDFS.
\item<2-> Emitir los pares clave/valor calculados por los mapeadores.
\item<3-> Particionar, transferir y ordenar los pares para agruparlos por clave.
\item<4-> Agregar cada grupo y escribir los pares finales.
\end{enumerate}
\end{frame}

\begin{columns}[T,onlytextwidth]
