\begin{frame}[fragile]{OpenMP: formar un equipo y repartir trabajo}{Repaso · Memoria compartida y modelo fork--join}
\begin{columns}[T,onlytextwidth]
\column{.57\textwidth}
\begin{itemize}[<+->]\setlength\itemsep{4pt}
  \item API para C, C++ y Fortran: directivas, rutinas y variables de entorno.
  \item \code{parallel} forma un equipo; cada hilo ejecuta el bloque.
  \item \code{for} reparte iteraciones entre los hilos del equipo.
  \item Al terminar la región paralela hay una barrera implícita.
\end{itemize}
\begin{uncoverenv}<3->
\begin{lstlisting}[style=compact]
#pragma omp parallel
{
    #pragma omp for
    for (int i = 0; i < n; ++i) procesar(i);
}
\end{lstlisting}

\end{block}
\end{columns}

\begin{frame}{Ejemplo intermedio 3: quicksort}{Dividir para conquistar · Dos subarreglos independientes}

% 24.
\begin{frame}{Por qué usar tareas para la recursión}{Un equipo de hilos ejecuta muchas unidades de trabajo}
\begin{columns}[T,onlytextwidth]
\column{.49\textwidth}
\begin{block}{Crear trabajo con task}
El hilo genera una tarea. Puede ejecutarla de inmediato o dejarla disponible para otro hilo del mismo equipo.
\end{block}
\begin{block}<2->{Coordinar con taskwait}
La tarea actual espera a sus hijas ya creadas. En quicksort, cada hija espera a sus propias hijas antes de terminar.
\end{block}
\column{.47\textwidth}
\begin{itemize}\setlength\itemsep{8pt}
\item<3-> Crear una sola región \code{parallel} al inicio.
\item<4-> Usar \code{single} para iniciar una única recursión.
\item<5-> Copiar límites y puntero con \code{firstprivate}.
\item<6-> Usar un umbral para no generar tareas diminutas.
\end{itemize}
\end{columns}
\uncover<6->{\smallnote{El original crea parallel sections en cada llamada: puede serializar niveles internos o crear demasiados equipos. Las tareas evitan ese patrón.}}
\end{frame}

\begin{frame}{Qué aporta el paso de mensajes}{Ventajas y costos que debemos considerar}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{block}{Ventajas}
\begin{itemize}\setlength\itemsep{6pt}
\item<1-> Se adapta a nodos conectados por una red.
\item<2-> Expresa particiones de datos de forma explícita.
\item<3-> Hace visible quién posee y comunica cada dato.
\item<4-> Permite usar memoria y cómputo de varios nodos.
\end{itemize}
\end{block}
\column{.48\textwidth}
\begin{block}{Costos y límites}
\begin{itemize}\setlength\itemsep{6pt}
\item<1-> La red añade latencia y limita el ancho de banda.
\item<2-> Hay que equilibrar trabajo y comunicaciones.
\item<3-> Un mensaje sin recepción compatible puede bloquear el programa.
\item<4-> Espacios separados no eliminan errores de sincronización.
\end{itemize}
\end{block}
\end{columns}
\end{frame}

\begin{frame}{MPI es una especificación de una interfaz}{Las implementaciones proporcionan bibliotecas y herramientas}
\begin{itemize}[<+->]\setlength\itemsep{10pt}
\item No es un lenguaje: se usa desde programas C, C++ o Fortran mediante sus interfaces disponibles.
\item El estándar define las operaciones y su comportamiento. Open MPI y MPICH son implementaciones.
\item \code{mpicc} facilita compilar y enlazar un programa C con la biblioteca MPI instalada.
\item \code{mpirun -np 4} inicia cuatro procesos en el entorno configurado.
\end{itemize}
\begin{block}<5->{Modelo SPMD}
Los procesos ejecutan el mismo programa, pero pueden trabajar con datos distintos y tomar ramas distintas según su identificador o \emph{rank}.
\end{block}
\uncover<5->{\smallnote{Aquí estudiamos comunicación de dos lados y colectivas. MPI también ofrece otras capacidades, como operaciones de acceso remoto a memoria.}}
\end{frame}

\begin{frame}[fragile]{Anatomía de un mensaje}{Comunicación punto a punto de dos lados}
\begin{lstlisting}
MPI_Send(buffer, count, datatype, dest, tag, comm);
MPI_Recv(buffer, capacity, datatype, source, tag, comm, status);
\end{lstlisting}
\begin{columns}[T,onlytextwidth]
\column{.49\textwidth}
\small
\begin{itemize}\setlength\itemsep{6pt}
\item<2-> \code{buffer}: dirección de los datos locales.
\item<3-> \code{count}: cantidad de elementos del tipo indicado, no necesariamente bytes.
\item<4-> \code{capacity}: espacio disponible en la recepción.
\end{itemize}
\column{.47\textwidth}
\small
\begin{itemize}\setlength\itemsep{6pt}
\item<2-> \code{dest/source}: rank del otro proceso.
\item<3-> \code{tag}: etiqueta entera para distinguir mensajes.
\item<4-> \code{comm}: contexto de comunicación; \code{status}: información del mensaje recibido.
\end{itemize}
\end{columns}
\begin{block}<5->{Compatibilidad}
La recepción debe coincidir en comunicador, fuente y etiqueta, y tener tipos compatibles y espacio suficiente. Usaremos fuentes y etiquetas explícitas.
\end{block}
\end{frame}

\begin{frame}{El orden de las comunicaciones importa}{Evitar esperas circulares}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{block}{Patrón que puede bloquearse}
Proceso 0: enviar a 1; luego recibir de 1.\\[6pt]
Proceso 1: enviar a 0; luego recibir de 0.
\end{block}
\uncover<2->{\small \code{MPI_Send} puede esperar una recepción compatible. Que funcione con mensajes pequeños no demuestra que el patrón sea correcto.}
\column{.48\textwidth}
\begin{block}<3->{Alternativas}
Ordenar: un proceso envía primero y el otro recibe primero.\\[6pt]
Usar \code{MPI_Sendrecv} para un intercambio.\\[6pt]
Publicar recepciones y envíos no bloqueantes y esperar su finalización.
\end{block}
\end{columns}
\par\vspace{4mm}
\uncover<4->{\small\textbf{Precisión:} al retornar \code{MPI_Send}, el buffer de envío puede reutilizarse; eso no implica que el receptor ya haya terminado de procesar los datos.}
\end{frame}

Operación & Esquema & Propósito \\

\leavevmode\uncover<2->{\code{MPI_Scatter}} & \leavevmode\uncover<2->{\((a,b,c)\rightarrow a\mid b\mid c\)} & \leavevmode\uncover<2->{Distribuir bloques desde la raíz.} \\

\leavevmode\uncover<4->{\code{MPI_Reduce}} & \leavevmode\uncover<4->{\(a\mid b\mid c\rightarrow a+b+c\)} & \leavevmode\uncover<4->{Combinar y entregar en la raíz.} \\
\leavevmode\uncover<5->{\code{MPI_Allreduce}} & \leavevmode\uncover<5->{\(a\mid b\mid c\rightarrow(s,s,s)\)} & \leavevmode\uncover<5->{Combinar y entregar a todos.} \\

\begin{block}<7->{Reglas para nuestros ejemplos con MPI\_COMM\_WORLD}
Todos llaman a las colectivas en el mismo orden y con argumentos compatibles. Una colectiva como \code{MPI_Bcast} no sustituye automáticamente una barrera.
\end{block}
\end{frame}
