\begin{frame}[fragile]{Comunicación punto a punto}{El envío y la recepción deben coincidir}
\begin{itemize}\setlength\itemsep{5pt}

\item El emisor ejecuta \code{MPI_Send}; el receptor ejecuta \code{MPI_Recv}.

MPI_Send(buffer, count, datatype, destination, tag, comm);
MPI_Recv(buffer, maxcount, datatype, source, tag, comm, &status);

\begin{block}{Identificar el mensaje}
\code{tag}: etiqueta entera.\\[3pt]
\code{comm}: grupo de procesos y contexto.\\[3pt]
\code{source / destination}: rango en ese comunicador.
\end{block}
\column{.48\textwidth}
\begin{block}{Identificar los datos}
\code{buffer}: dirección de memoria.\\[3pt]
\code{count}: número de elementos.\\[3pt]
\code{datatype}: tipo MPI de los elementos.\\[3pt]
\code{status}: información de la recepción.
\end{block}
\end{columns}

\section{MPI: operaciones colectivas}
% 05. MPI p. 6.
\begin{frame}[fragile]{Colectivas: broadcast y reduce}{Participan todos los procesos del comunicador}
\begin{block}{Broadcast: difundir un mismo dato}
El proceso \code{root} comunica su buffer a todos, incluido él mismo.
\end{block}
\begin{lstlisting}
int MPI_Bcast(void *buffer, int count, MPI_Datatype datatype,
              int root, MPI_Comm comm);
\end{lstlisting}
\begin{block}{Reduce: combinar contribuciones}
Aplica \code{op} a los valores de todos los procesos. El resultado queda en \code{root}.
\end{block}
\begin{lstlisting}
int MPI_Reduce(const void *sendbuf, void *recvbuf, int count,
               MPI_Datatype datatype, MPI_Op op,
               int root, MPI_Comm comm);
\end{lstlisting}
\smallnote{Ejemplos de operación: \code{MPI_SUM}, \code{MPI_MAX}, \code{MPI_MIN}. Todos deben llamar a las colectivas en un orden compatible.}
\end{frame}

% 06. MPI p. 7.
\begin{frame}[fragile]{Colectivas: gather y scatter}{Reunir bloques o distribuir bloques}
\small
\begin{block}{Gather: reunir en root}
Cada proceso envía un bloque; \code{root} los recibe en orden de rango.
\end{block}
\begin{lstlisting}[style=compact]
int MPI_Gather(const void *sendbuf, int sendcount,
               MPI_Datatype sendtype, void *recvbuf,
               int recvcount, MPI_Datatype recvtype,
               int root, MPI_Comm comm);
\end{lstlisting}
\begin{block}{Scatter: repartir desde root}
\code{root} envía un bloque distinto a cada proceso.
\end{block}
\begin{lstlisting}[style=compact]
int MPI_Scatter(const void *sendbuf, int sendcount,
                MPI_Datatype sendtype, void *recvbuf,
                int recvcount, MPI_Datatype recvtype,
                int root, MPI_Comm comm);
\end{lstlisting}
\smallnote{Los conteos son por proceso: \code{recvcount} en Gather no es el tamaño total del buffer de root.}
\end{frame}

% 09. MPI p. 9. Formula y grafico nativos.
\begin{frame}{Aproximar el valor de \(\pi\)}{Integración numérica con la regla del punto medio}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\[\pi=\int_0^1\frac{4}{1+x^2}\,dx.\]
\small Dividimos \([0,1]\) en \(n\) intervalos y sumamos áreas de rectángulos:
\[h=\frac1n,\qquad x_i=h\left(i-\frac12\right).\]
\[\pi\approx h\sum_{i=1}^{n}\frac{4}{1+x_i^2}.\]
\begin{block}{Trabajo paralelo}
Cada proceso calcula una parte de la suma. Una reducción suma todas las contribuciones.
\end{block}
\column{.48\textwidth}\centering

  tick label style={font=\scriptsize},label style={font=\small},axis lines=left]
\foreach \a/\b/\h in {0/.2/3.9604,.2/.4/3.6697,.4/.6/3.2,.6/.8/2.6846,.8/1/2.2099}{

\end{axis}
\end{tikzpicture}
\smallnote{Ilustración con \(n=5\); el programa usa \(n=5000\).}
\end{columns}
\end{frame}

\section{Modelo híbrido: MPI y OpenMP / Pthreads}
% 19. Hibrid p. 4.
\begin{frame}{Procesos e hilos}{MPI organiza procesos; OpenMP y Pthreads organizan hilos}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{block}{Proceso}
Tiene su propio espacio de direcciones. Un programa MPI se ejecuta como varios procesos,
cada uno identificado por su rango.
\end{block}
\begin{itemize}\setlength\itemsep{6pt}
\item Un proceso puede crear uno o más hilos.
\item Al entrar a \code{main}, cada proceso tiene un hilo inicial de aplicación.
\end{itemize}
\column{.48\textwidth}
\begin{block}{Hilo}
Comparte con los hilos del proceso la memoria y recursos como los descriptores de archivo.
\end{block}
\begin{itemize}\setlength\itemsep{6pt}
\item Puede leer y escribir las mismas direcciones.
\item Tiene su propio contador de instrucciones y pila.
\item Puede mantener variables privadas en su pila.
\end{itemize}
\end{columns}
\begin{block}{Cuidar los datos compartidos}
Compartir memoria permite comunicación local rápida; las escrituras concurrentes deben coordinarse.
\end{block}
\end{frame}

% 21. Hibrid p. 6. Ventajas presentadas como posibilidades, no garantias.
\begin{frame}{Ventajas del modelo híbrido}{Adaptar el reparto de trabajo al hardware del clúster}
\begin{itemize}\setlength\itemsep{5pt}
\item \textbf{Escalabilidad:} MPI distribuye el trabajo entre servidores; los hilos aprovechan los núcleos de cada nodo.
\item \textbf{Rendimiento y flexibilidad:} se puede elegir cuántos procesos e hilos usar y cómo repartir las tareas.
\item \textbf{Uso de recursos:} los hilos comparten buffers locales y aprovechan las CPU del nodo.
\item \textbf{Comunicación:} menos procesos MPI pueden reducir el número de mensajes y algunos costos de comunicación.
\item \textbf{Organización del programa:} MPI gestiona el nivel distribuido; OpenMP o Pthreads gestiona el nivel local.
\end{itemize}
\begin{block}{El beneficio se comprueba midiendo}
La mejora depende del algoritmo, del equilibrio de carga, de la memoria y de la configuración de procesos e hilos.
\end{block}
\end{frame}

% 22. Hibrid p. 7. FUNNELED: hilo que inicializa MPI.
\begin{frame}{Niveles de soporte de hilos en MPI}{Los hilos del mismo proceso comparten los objetos MPI}
\small\renewcommand{\arraystretch}{1.55}
\begin{tabularx}{\textwidth}{lX}
\toprule Nivel solicitado & Regla de uso \\
\midrule
\code{MPI_THREAD_SINGLE} & Solo se ejecuta un hilo. \\
\code{MPI_THREAD_FUNNELED} & Hay varios hilos; solo el hilo que inicializó MPI hace llamadas MPI. \\
\code{MPI_THREAD_SERIALIZED} & Varios hilos pueden llamar MPI, pero las llamadas se serializan: nunca al mismo tiempo. \\
\code{MPI_THREAD_MULTIPLE} & Varios hilos pueden hacer llamadas MPI simultáneamente. \\
\bottomrule
\end{tabularx}
\begin{block}{Solicitar y verificar}
Se usa \code{MPI_Init_thread} y se comprueba \code{provided}.
Solicitar un nivel no garantiza que la implementación lo proporcione.
\end{block}
\smallnote{Referencia: \href{https://docs.open-mpi.org/en/v5.0.0/man-openmpi/man3/MPI_Init_thread.3.html}{Open MPI: MPI\_Init\_thread}.}
\end{frame}

\begin{frame}[fragile]{Medir tiempo con OpenMP y MPI}{Comparar tiempos transcurridos en el mismo hilo o proceso}
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}
\begin{block}{OpenMP: tiempo local}
\code{omp_get_wtime()} devuelve segundos.
\end{block}
\begin{lstlisting}[style=compact]
double t1 = omp_get_wtime();
// Computo costoso, incluido el join.
double t2 = omp_get_wtime();
printf("Tiempo local: %g s\n", t2 - t1);
\end{lstlisting}
\column{.49\textwidth}
\begin{block}{MPI: duración del proceso más lento}
\code{MPI_Wtime()} devuelve segundos.
\end{block}
\begin{lstlisting}[style=compact]
MPI_Barrier(MPI_COMM_WORLD);
double t1 = MPI_Wtime();
// Computo y comunicaciones a medir.
double elapsed = MPI_Wtime() - t1;
double maximum;
MPI_Reduce(&elapsed, &maximum, 1,
           MPI_DOUBLE, MPI_MAX, 0,
           MPI_COMM_WORLD);
if (rank == 0)
    printf("Tiempo maximo: %g s\n", maximum);
\end{lstlisting}
\end{columns}
\smallnote{No restar marcas de tiempo tomadas en procesos distintos: sus relojes no tienen por qué estar sincronizados. La reducción final queda fuera de la región medida.}
\end{frame}
